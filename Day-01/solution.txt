from collections import Counter
for _ in range(int(input())):    
    n = int(input())
    a = list(map(int,input().split()))
    c = Counter(a)
    print(c)
    ans = 0
    for j in c.values():
        temp =0
        for i in c.keys():
            if c[i]>=j:
                temp+=j
                print(temp)
        ans = max(temp,ans)
    print(ans)
